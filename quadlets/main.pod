# The pod groups all seven containers into one network namespace, so they reach each other
# on localhost, and publishes their ports. Which ports comes from runtime.env: quadlet
# copies these lines verbatim into the generated unit, where systemd expands them against
# the EnvironmentFile -- which is why the pod needs one although it runs no container.
#
# The web and database ports are the host side only; inside the pod nginx and PostgreSQL
# keep 80, 443 and 5432. The dropzone endpoints publish the port they listen on, so
# SFTP_PORT and FLIGHT_PORT are one setting each.
#
# PostgreSQL is published for reporting and the tools that write bronze, skipping the proxy
# because its wire protocol is not HTTP; its own passwords guard it.
#
# The file is named main.pod, so the generated unit is main-pod.service and containers link
# to it with Pod=main.pod, while PodName below is what podman shows.
[Pod]
PodName=${APP_NAME}
PublishPort=${HTTP_PORT}:80
PublishPort=${HTTPS_PORT}:443
PublishPort=${PG_PORT}:5432
PublishPort=${SFTP_PORT}:${SFTP_PORT}
PublishPort=${FLIGHT_PORT}:${FLIGHT_PORT}

# Read by systemd, not podman: this unit starts only the pod and passes nothing to
# --env-file. A missing file fails it, as an unexpanded port would a moment later.
[Service]
EnvironmentFile=%h/.config/${APP_NAME}/runtime.env

[Install]
WantedBy=default.target
